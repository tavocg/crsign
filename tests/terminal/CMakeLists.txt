cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(crsign_terminal_tests LANGUAGES C)
    enable_testing()
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_executable(terminal_driver terminal_driver.c ../../sign/terminal.c)
target_include_directories(terminal_driver PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set_target_properties(terminal_driver PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(terminal_driver PRIVATE /W4 /WX)
else()
    target_compile_options(terminal_driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(UNIX)
    add_test(NAME terminal_pin COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/terminal_test.py" $<TARGET_FILE:terminal_driver>)
    set_tests_properties(terminal_pin PROPERTIES TIMEOUT 30)
endif()
